"""E3-P1 pure-test runner: no conftest, no network, and no DB/ORM/app modules."""
import importlib.abc
import socket
import sys

import pytest

sys.stdout.reconfigure(encoding="utf-8")


def _no_network(*args, **kwargs):
    raise AssertionError("Network forbidden in pure E3-P1 tests")


socket.socket.connect = _no_network
FORBIDDEN = ("sqlalchemy", "psycopg2", "fastapi", "pydantic_settings", "app.main",
             "app.core", "app.models", "app.api", "requests", "httpx")


def forbidden(name):
    return any(name == b or name.startswith(b + ".") for b in FORBIDDEN) or name.endswith(".conftest")


class Guard(importlib.abc.MetaPathFinder):
    def find_spec(self, fullname, path=None, target=None):
        if forbidden(fullname):
            raise AssertionError("Forbidden module: " + fullname)


sys.meta_path.insert(0, Guard())
result = pytest.main(["--noconftest", "-p", "no:cacheprovider", *sys.argv[1:]])
loaded = sorted(n for n in sys.modules if forbidden(n))
print("Forbidden modules loaded:", loaded)
assert not loaded
raise SystemExit(result)

    original = parser.parse_artifact

    def patched(artifact, **kw):

    def unsorted(value):
        canonical._check(value)

    canonical.canonical_json = unsorted
    parser.canonical_json = unsorted
elif mutant == "no_duplicate_key_flag":
    original_finalize = parser._finalize

    def finalize(drafts):

        return original_finalize(drafts)
    parser._finalize = finalize
